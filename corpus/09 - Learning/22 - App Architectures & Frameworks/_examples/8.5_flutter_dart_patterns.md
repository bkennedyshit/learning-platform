## 1. Stateful Widget with Lifecycle

```dart
import 'package:flutter/material.dart';

/// Demonstrates the complete StatefulWidget lifecycle.
/// Each lifecycle method is annotated with when and why it's called.
class LifecycleDemo extends StatefulWidget {
  final String title;

  // const constructor: allows Flutter to optimize by reusing instances
  const LifecycleDemo({super.key, required this.title});

  @override
  State<LifecycleDemo> createState() => _LifecycleDemoState();
  // createState() is called ONCE when the framework first inserts
  // this widget into the tree. The State object persists across rebuilds.
}

class _LifecycleDemoState extends State<LifecycleDemo>
    with WidgetsBindingObserver {
  // WidgetsBindingObserver: notifies us of app lifecycle changes
  // (background, foreground, inactive)

  late final TextEditingController _controller;
  int _rebuildCount = 0;
  AppLifecycleState? _appState;

  @override
  void initState() {
    super.initState();
    // Called ONCE after createState(). Use for:
    // - Creating controllers, animation controllers
    // - Subscribing to streams
    // - One-time initialization
    // DO NOT call setState() here (widget hasn't built yet).
    _controller = TextEditingController(text: 'Hello');
    WidgetsBinding.instance.addObserver(this);
    debugPrint('initState called');
  }

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    // Called after initState() AND whenever an InheritedWidget
    // that this widget depends on changes (e.g., Theme, MediaQuery).
    // Safe to call setState() here.
    debugPrint('didChangeDependencies called');
  }

  @override
  void didUpdateWidget(covariant LifecycleDemo oldWidget) {
    super.didUpdateWidget(oldWidget);
    // Called when the PARENT rebuilds and provides a new widget instance
    // with the same runtimeType and key. Use to react to prop changes.
    if (oldWidget.title != widget.title) {
      debugPrint('Title changed: ${oldWidget.title} → ${widget.title}');
    }
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    // Called when app goes to background/foreground
    setState(() => _appState = state);
    debugPrint('App lifecycle: $state');
  }

  @override
  Widget build(BuildContext context) {
    _rebuildCount++;
    // build() is called whenever:
    // 1. setState() is called on this State
    // 2. Parent rebuilds with new widget config
    // 3. An InheritedWidget dependency changes
    // Must be PURE: no side effects, no async calls.
    return Card(
      margin: const EdgeInsets.all(16),
      child: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          mainAxisSize: MainAxisSize.min,
          children: [
            Text('Title: ${widget.title}',
                style: Theme.of(context).textTheme.titleLarge),
            Text('Rebuild count: $_rebuildCount'),
            Text('App state: ${_appState ?? "active"}'),
            const SizedBox(height: 8),
            TextField(
              controller: _controller,
              decoration: const InputDecoration(labelText: 'Type here'),
            ),
            const SizedBox(height: 8),
            ElevatedButton(
              onPressed: () => setState(() {}), // Force rebuild
              child: const Text('Force Rebuild'),
            ),
          ],
        ),
      ),
    );
  }

  @override
  void deactivate() {
    super.deactivate();
    // Called when the element is removed from the tree (temporarily).
    // The State might be reinserted later (e.g., with GlobalKey).
    debugPrint('deactivate called');
  }

  @override
  void dispose() {
    // Called when the State is permanently removed. Use for:
    // - Disposing controllers (TextEditingController, AnimationController)
    // - Cancelling subscriptions
    // - Removing observers
    // After dispose(), setState() must NEVER be called.
    _controller.dispose();
    WidgetsBinding.instance.removeObserver(this);
    debugPrint('dispose called');
    super.dispose();
  }
}
```

## 2. Riverpod Provider (Async Data + Caching)

```dart
import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:http/http.dart' as http;

// ─── Data Layer ──────────────────────────────────────────────────

class Post {
  final int id;
  final String title;
  final String body;

  Post({required this.id, required this.title, required this.body});

  factory Post.fromJson(Map<String, dynamic> json) => Post(
        id: json['id'],
        title: json['title'],
        body: json['body'],
      );
}

// ─── Providers ───────────────────────────────────────────────────

// FutureProvider: automatically handles loading/error/data states.
// autoDispose: cleans up when no widget is listening (prevents memory leaks).
// family: parameterized provider (different instance per userId).
final userPostsProvider =
    FutureProvider.autoDispose.family<List<Post>, int>((ref, userId) async {
  // ref.onDispose: called when this provider is disposed (cleanup)
  final cancelToken = CancelToken();
  ref.onDispose(() => cancelToken.cancel());

  final response = await http.get(
    Uri.parse('https://jsonplaceholder.typicode.com/posts?userId=$userId'),
  );

  if (response.statusCode != 200) {
    throw Exception('Failed to load posts: ${response.statusCode}');
  }

  final data = jsonDecode(response.body) as List;
  return data.map((json) => Post.fromJson(json)).toList();
});

// StateProvider: simple mutable state (like useState in React)
final selectedUserIdProvider = StateProvider<int>((ref) => 1);

// ─── UI Layer ────────────────────────────────────────────────────

class PostsPage extends ConsumerWidget {
  const PostsPage({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final userId = ref.watch(selectedUserIdProvider);
    // watch: subscribes to the provider. Widget rebuilds when value changes.
    final postsAsync = ref.watch(userPostsProvider(userId));

    return Scaffold(
      appBar: AppBar(
        title: Text('Posts for User $userId'),
        actions: [
          // Dropdown to switch users
          DropdownButton<int>(
            value: userId,
            items: List.generate(10, (i) => i + 1)
                .map((id) => DropdownMenuItem(value: id, child: Text('User $id')))
                .toList(),
            onChanged: (id) {
              if (id != null) {
                // read: access provider without subscribing (for mutations)
                ref.read(selectedUserIdProvider.notifier).state = id;
              }
            },
          ),
        ],
      ),
      body: postsAsync.when(
        // AsyncValue.when: exhaustive pattern matching on loading/error/data
        loading: () => const Center(child: CircularProgressIndicator()),
        error: (error, stack) => Center(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Text('Error: $error'),
              ElevatedButton(
                // invalidate: forces the provider to re-fetch
                onPressed: () => ref.invalidate(userPostsProvider(userId)),
                child: const Text('Retry'),
              ),
            ],
          ),
        ),
        data: (posts) => RefreshIndicator(
          // Pull-to-refresh: invalidates the provider to re-fetch
          onRefresh: () => ref.refresh(userPostsProvider(userId).future),
          child: ListView.builder(
            itemCount: posts.length,
            itemBuilder: (context, index) {
              final post = posts[index];
              return ListTile(
                title: Text(post.title),
                subtitle: Text(post.body, maxLines: 2, overflow: TextOverflow.ellipsis),
              );
            },
          ),
        ),
      ),
    );
  }
}
```

## 3. BLoC with Repository Pattern

```dart
import 'package:flutter/material.dart';
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:equatable/equatable.dart';

// ─── Domain Layer ────────────────────────────────────────────────

class Todo {
  final String id;
  final String title;
  final bool completed;

  const Todo({required this.id, required this.title, this.completed = false});

  Todo copyWith({String? title, bool? completed}) =>
      Todo(id: id, title: title ?? this.title, completed: completed ?? this.completed);
}

abstract class TodoRepository {
  Future<List<Todo>> fetchAll();
  Future<void> add(Todo todo);
  Future<void> toggle(String id);
  Future<void> delete(String id);
}

// ─── BLoC Events ─────────────────────────────────────────────────

sealed class TodoEvent extends Equatable {
  @override
  List<Object?> get props => [];
}

class TodosLoaded extends TodoEvent {}
class TodoAdded extends TodoEvent {
  final String title;
  TodoAdded(this.title);
  @override
  List<Object?> get props => [title];
}
class TodoToggled extends TodoEvent {
  final String id;
  TodoToggled(this.id);
  @override
  List<Object?> get props => [id];
}
class TodoDeleted extends TodoEvent {
  final String id;
  TodoDeleted(this.id);
  @override
  List<Object?> get props => [id];
}

// ─── BLoC States ─────────────────────────────────────────────────

sealed class TodoState extends Equatable {
  @override
  List<Object?> get props => [];
}

class TodoInitial extends TodoState {}
class TodoLoading extends TodoState {}
class TodoSuccess extends TodoState {
  final List<Todo> todos;
  TodoSuccess(this.todos);
  @override
  List<Object?> get props => [todos];
}
class TodoFailure extends TodoState {
  final String message;
  TodoFailure(this.message);
  @override
  List<Object?> get props => [message];
}

// ─── BLoC Implementation ─────────────────────────────────────────

class TodoBloc extends Bloc<TodoEvent, TodoState> {
  final TodoRepository _repository;

  TodoBloc({required TodoRepository repository})
      : _repository = repository,
        super(TodoInitial()) {
    on<TodosLoaded>(_onLoaded);
    on<TodoAdded>(_onAdded);
    on<TodoToggled>(_onToggled);
    on<TodoDeleted>(_onDeleted);
  }

  Future<void> _onLoaded(TodosLoaded event, Emitter<TodoState> emit) async {
    emit(TodoLoading());
    try {
      final todos = await _repository.fetchAll();
      emit(TodoSuccess(todos));
    } catch (e) {
      emit(TodoFailure(e.toString()));
    }
  }

  Future<void> _onAdded(TodoAdded event, Emitter<TodoState> emit) async {
    final todo = Todo(id: DateTime.now().toIso8601String(), title: event.title);
    await _repository.add(todo);
    add(TodosLoaded()); // Reload list
  }

  Future<void> _onToggled(TodoToggled event, Emitter<TodoState> emit) async {
    await _repository.toggle(event.id);
    add(TodosLoaded());
  }

  Future<void> _onDeleted(TodoDeleted event, Emitter<TodoState> emit) async {
    await _repository.delete(event.id);
    add(TodosLoaded());
  }
}

// ─── UI ──────────────────────────────────────────────────────────

class TodoPage extends StatelessWidget {
  const TodoPage({super.key});

  @override
  Widget build(BuildContext context) {
    return BlocProvider(
      create: (context) => TodoBloc(
        repository: context.read<TodoRepository>(),
      )..add(TodosLoaded()), // Load on creation
      child: const TodoView(),
    );
  }
}

class TodoView extends StatelessWidget {
  const TodoView({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Todos (BLoC)')),
      body: BlocBuilder<TodoBloc, TodoState>(
        builder: (context, state) => switch (state) {
          TodoInitial() || TodoLoading() =>
            const Center(child: CircularProgressIndicator()),
          TodoFailure(:final message) =>
            Center(child: Text('Error: $message')),
          TodoSuccess(:final todos) => ListView.builder(
            itemCount: todos.length,
            itemBuilder: (context, index) {
              final todo = todos[index];
              return Dismissible(
                key: ValueKey(todo.id),
                onDismissed: (_) =>
                    context.read<TodoBloc>().add(TodoDeleted(todo.id)),
                child: CheckboxListTile(
                  value: todo.completed,
                  title: Text(todo.title),
                  onChanged: (_) =>
                      context.read<TodoBloc>().add(TodoToggled(todo.id)),
                ),
              );
            },
          ),
        },
      ),
    );
  }
}
```

## 4. CustomPainter (Interactive Chart)

```dart
import 'dart:math';
import 'package:flutter/material.dart';

/// A bar chart widget built entirely with CustomPainter.
/// Demonstrates: Canvas API, hit testing, animations, responsive sizing.
class BarChart extends StatefulWidget {
  final List<double> data;
  final List<String> labels;

  const BarChart({super.key, required this.data, required this.labels});

  @override
  State<BarChart> createState() => _BarChartState();
}

class _BarChartState extends State<BarChart>
    with SingleTickerProviderStateMixin {
  late AnimationController _animController;
  int? _hoveredIndex;

  @override
  void initState() {
    super.initState();
    _animController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 800),
    )..forward(); // Animate bars growing on first render
  }

  @override
  void dispose() {
    _animController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return GestureDetector(
      onTapDown: (details) {
        // Hit test: determine which bar was tapped
        final barWidth = context.size!.width / widget.data.length;
        final index = (details.localPosition.dx / barWidth).floor();
        if (index >= 0 && index < widget.data.length) {
          setState(() => _hoveredIndex = index);
        }
      },
      onTapUp: (_) => setState(() => _hoveredIndex = null),
      child: AnimatedBuilder(
        animation: _animController,
        builder: (context, _) => CustomPaint(
          painter: _BarChartPainter(
            data: widget.data,
            labels: widget.labels,
            animationProgress: _animController.value,
            hoveredIndex: _hoveredIndex,
            barColor: Theme.of(context).colorScheme.primary,
            labelColor: Theme.of(context).colorScheme.onSurface,
          ),
          size: const Size(double.infinity, 250),
        ),
      ),
    );
  }
}

class _BarChartPainter extends CustomPainter {
  final List<double> data;
  final List<String> labels;
  final double animationProgress;
  final int? hoveredIndex;
  final Color barColor;
  final Color labelColor;

  _BarChartPainter({
    required this.data,
    required this.labels,
    required this.animationProgress,
    required this.hoveredIndex,
    required this.barColor,
    required this.labelColor,
  });

  @override
  void paint(Canvas canvas, Size size) {
    if (data.isEmpty) return;

    final maxValue = data.reduce(max);
    final barWidth = size.width / data.length;
    final chartHeight = size.height - 30; // Reserve space for labels
    final padding = barWidth * 0.2;

    for (int i = 0; i < data.length; i++) {
      final normalizedHeight = (data[i] / maxValue) * chartHeight;
      // Animation: bars grow from bottom
      final animatedHeight = normalizedHeight * animationProgress;

      final isHovered = i == hoveredIndex;
      final paint = Paint()
        ..color = isHovered ? barColor : barColor.withOpacity(0.7)
        ..style = PaintingStyle.fill;

      // Bar rectangle
      final rect = Rect.fromLTWH(
        i * barWidth + padding,
        chartHeight - animatedHeight,
        barWidth - padding * 2,
        animatedHeight,
      );

      // Rounded top corners
      canvas.drawRRect(
        RRect.fromRectAndCorners(rect,
            topLeft: const Radius.circular(4),
            topRight: const Radius.circular(4)),
        paint,
      );

      // Value label above bar
      if (animationProgress > 0.8) {
        final textPainter = TextPainter(
          text: TextSpan(
            text: data[i].toStringAsFixed(0),
            style: TextStyle(color: labelColor, fontSize: 11),
          ),
          textDirection: TextDirection.ltr,
        )..layout();
        textPainter.paint(
          canvas,
          Offset(
            i * barWidth + (barWidth - textPainter.width) / 2,
            chartHeight - animatedHeight - 16,
          ),
        );
      }

      // X-axis label
      final labelPainter = TextPainter(
        text: TextSpan(
          text: labels[i],
          style: TextStyle(color: labelColor.withOpacity(0.7), fontSize: 10),
        ),
        textDirection: TextDirection.ltr,
      )..layout();
      labelPainter.paint(
        canvas,
        Offset(i * barWidth + (barWidth - labelPainter.width) / 2, chartHeight + 8),
      );
    }
  }

  @override
  bool shouldRepaint(_BarChartPainter old) =>
      old.animationProgress != animationProgress ||
      old.hoveredIndex != hoveredIndex;
}
```

## 5. Isolate Compute (Image Processing)

```dart
import 'dart:isolate';
import 'dart:typed_data';
import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';

/// Demonstrates offloading heavy computation to a separate Isolate
/// while keeping the UI responsive (animation keeps running).
class IsolateDemo extends StatefulWidget {
  const IsolateDemo({super.key});

  @override
  State<IsolateDemo> createState() => _IsolateDemoState();
}

class _IsolateDemoState extends State<IsolateDemo> {
  String _status = 'Tap to start computation';
  double _progress = 0;
  bool _computing = false;

  Future<void> _runComputation() async {
    setState(() {
      _computing = true;
      _status = 'Computing on separate isolate...';
    });

    // compute() is Flutter's convenience wrapper around Isolate.spawn.
    // It spawns an isolate, runs the function, and returns the result.
    // The function must be a TOP-LEVEL or STATIC function.
    final result = await compute(_heavyMathComputation, 50000000);

    setState(() {
      _computing = false;
      _status = 'Result: ${result.toStringAsFixed(6)}';
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Isolate Demo')),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            // This animation proves the UI stays responsive
            // even during heavy computation on the worker isolate.
            const SizedBox(
              width: 100,
              height: 100,
              child: CircularProgressIndicator(strokeWidth: 6),
            ),
            const SizedBox(height: 32),
            Text(_status, textAlign: TextAlign.center),
            const SizedBox(height: 16),
            ElevatedButton(
              onPressed: _computing ? null : _runComputation,
              child: Text(_computing ? 'Computing...' : 'Start Heavy Computation'),
            ),
            const SizedBox(height: 32),
            const Text(
              'The spinner above keeps animating smoothly\n'
              'because computation runs on a separate isolate.',
              textAlign: TextAlign.center,
              style: TextStyle(color: Colors.grey),
            ),
          ],
        ),
      ),
    );
  }
}

// TOP-LEVEL FUNCTION: required for compute().
// Cannot be a method, closure, or lambda.
// Cannot access any instance variables or widgets.
double _heavyMathComputation(int iterations) {
  // Simulate heavy work: compute partial sum of harmonic series
  double sum = 0;
  for (int i = 1; i <= iterations; i++) {
    sum += 1.0 / i;
    // Additional work to make it genuinely slow
    if (i % 1000 == 0) {
      sum += (sum * 0.0001).abs();
    }
  }
  return sum;
}
```

## 6. FFI Bridge (Calling Native C Code)

```dart
// Demonstrates calling a native C function from Dart using FFI.
// This is how you integrate high-performance native libraries
// (image processing, crypto, physics engines) into Flutter.

import 'dart:ffi';
import 'dart:io';
import 'package:ffi/package:ffi.dart';

// ─── Step 1: Define the C function signature ─────────────────────
// Suppose we have a C library with:
// int32_t add_numbers(int32_t a, int32_t b);
// double compute_distance(double x1, double y1, double x2, double y2);

// Dart representation of the C function signature
typedef AddNumbersC = Int32 Function(Int32 a, Int32 b);
// Dart function type for calling it
typedef AddNumbersDart = int Function(int a, int b);

typedef ComputeDistanceC = Double Function(
    Double x1, Double y1, Double x2, Double y2);
typedef ComputeDistanceDart = double Function(
    double x1, double y1, double x2, double y2);

// ─── Step 2: Load the native library ─────────────────────────────
class NativeMath {
  late final DynamicLibrary _lib;
  late final AddNumbersDart addNumbers;
  late final ComputeDistanceDart computeDistance;

  NativeMath() {
    // Load the compiled shared library (.so on Linux/Android, .dylib on macOS, .dll on Windows)
    _lib = _loadLibrary();

    // Look up the function by its C symbol name
    addNumbers = _lib
        .lookupFunction<AddNumbersC, AddNumbersDart>('add_numbers');

    computeDistance = _lib
        .lookupFunction<ComputeDistanceC, ComputeDistanceDart>('compute_distance');
  }

  DynamicLibrary _loadLibrary() {
    if (Platform.isAndroid || Platform.isLinux) {
      return DynamicLibrary.open('libnative_math.so');
    } else if (Platform.isMacOS) {
      return DynamicLibrary.open('libnative_math.dylib');
    } else if (Platform.isWindows) {
      return DynamicLibrary.open('native_math.dll');
    }
    throw UnsupportedError('Platform not supported');
  }
}

// ─── Step 3: Use in Flutter ──────────────────────────────────────
// final math = NativeMath();
// final sum = math.addNumbers(3, 4); // Returns 7 (computed in C)
// final dist = math.computeDistance(0, 0, 3, 4); // Returns 5.0

// ─── The C source (compile separately) ───────────────────────────
// native_math.c:
// #include <math.h>
// #include <stdint.h>
//
// int32_t add_numbers(int32_t a, int32_t b) {
//     return a + b;
// }
//
// double compute_distance(double x1, double y1, double x2, double y2) {
//     double dx = x2 - x1;
//     double dy = y2 - y1;
//     return sqrt(dx * dx + dy * dy);
// }
//
// Compile: gcc -shared -o libnative_math.so native_math.c -lm
```
