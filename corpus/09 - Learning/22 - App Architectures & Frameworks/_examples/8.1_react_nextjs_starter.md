## 1. App Router Page with Server Component

```tsx
// app/posts/page.tsx
// ─────────────────────────────────────────────────────────────────
// This is a SERVER COMPONENT by default (no 'use client' directive).
// It runs exclusively on the server during request time or build time.
// Its JavaScript is never included in the client bundle.

// You can import server-only utilities directly — they never leak to the client.
import { db } from '@/lib/database';

// Metadata export: Next.js uses this to generate <head> tags.
// This is a static metadata object; for dynamic metadata, export generateMetadata().
export const metadata = {
  title: 'All Posts',
  description: 'Browse our latest articles',
};

// The component is async — you can await directly in the function body.
// No useEffect, no loading state management, no client-side fetch.
export default async function PostsPage() {
  // Direct database query — this runs on the server.
  // The SQL/ORM call happens at request time (SSR) or build time (SSG).
  const posts = await db.post.findMany({
    orderBy: { createdAt: 'desc' },
    take: 20,
    select: { id: true, title: true, excerpt: true, createdAt: true },
  });

  return (
    <main className="max-w-3xl mx-auto p-6">
      <h1 className="text-3xl font-bold mb-8">All Posts</h1>

      {/* Static list rendered on server — zero client JS for this section */}
      <ul className="space-y-4">
        {posts.map((post) => (
          <li key={post.id} className="border-b pb-4">
            <h2 className="text-xl font-semibold">{post.title}</h2>
            <p className="text-gray-600 mt-1">{post.excerpt}</p>
            <time className="text-sm text-gray-400">
              {new Date(post.createdAt).toLocaleDateString()}
            </time>
          </li>
        ))}
      </ul>

      {/* Client Component boundary: SearchFilter ships JS to the browser */}
      <SearchFilter />
    </main>
  );
}
```

**Key takeaways:**
- No `'use client'` → runs on server only, zero JS shipped for this component.
- `async/await` directly in component body — impossible in Client Components.
- Database imports stay server-side — no API route needed for read operations.
- Child Client Components (`<SearchFilter />`) hydrate independently.

## 2. Client Component with useEffect Cleanup

```tsx
// app/components/SearchFilter.tsx
'use client';
// ─────────────────────────────────────────────────────────────────
// 'use client' marks the SERIALIZATION BOUNDARY.
// This component and all its imports are included in the client JS bundle.
// It can use hooks, event handlers, and browser APIs.

import { useState, useEffect, useTransition } from 'react';

interface SearchResult {
  id: string;
  title: string;
  score: number;
}

export function SearchFilter() {
  // ─── State declarations ────────────────────────────────────────
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<SearchResult[]>([]);
  const [error, setError] = useState<string | null>(null);

  // useTransition: marks state updates as non-urgent.
  // isPending is true while the transition is processing.
  // React won't block user input while the transition runs.
  const [isPending, startTransition] = useTransition();

  // ─── Effect: fetch search results ─────────────────────────────
  useEffect(() => {
    // Guard: don't fetch for empty queries
    if (query.length < 2) {
      setResults([]);
      return; // No cleanup needed if we didn't start a fetch
    }

    // AbortController: cancels the fetch if query changes or component unmounts.
    // Each effect invocation creates its own controller instance.
    const controller = new AbortController();

    // Debounce via setTimeout inside the effect
    const timeoutId = setTimeout(async () => {
      try {
        const response = await fetch(
          `/api/search?q=${encodeURIComponent(query)}`,
          { signal: controller.signal }
        );

        if (!response.ok) throw new Error(`Search failed: ${response.status}`);

        const data: SearchResult[] = await response.json();

        // Wrap in startTransition: updating results is non-urgent.
        // If the user is still typing, React can interrupt this update.
        startTransition(() => {
          setResults(data);
          setError(null);
        });
      } catch (err) {
        if (err instanceof Error && err.name === 'AbortError') return;
        setError(err instanceof Error ? err.message : 'Unknown error');
      }
    }, 300); // 300ms debounce

    // ─── Cleanup function ──────────────────────────────────────
    // Runs when: (1) query changes, (2) component unmounts
    return () => {
      clearTimeout(timeoutId);   // Cancel pending debounce timer
      controller.abort();         // Cancel in-flight fetch request
    };
  }, [query]); // Dependency: re-run only when query changes

  return (
    <div className="mt-8">
      <input
        type="search"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        placeholder="Filter posts..."
        className="w-full p-3 border rounded-lg"
        aria-label="Search posts"
      />

      {/* Accessible loading indicator */}
      {isPending && <p aria-live="polite" className="text-sm mt-2">Searching...</p>}
      {error && <p role="alert" className="text-red-500 mt-2">{error}</p>}

      <ul className="mt-4 space-y-2">
        {results.map((result) => (
          <li key={result.id} className="p-2 bg-gray-50 rounded">
            {result.title}
            <span className="text-xs text-gray-400 ml-2">
              relevance: {result.score.toFixed(2)}
            </span>
          </li>
        ))}
      </ul>
    </div>
  );
}
```

## 3. Custom Hook with Generics

```tsx
// hooks/useLocalStorage.ts
'use client';
// ─────────────────────────────────────────────────────────────────
// A type-safe custom hook that syncs state with localStorage.
// Demonstrates: generics, lazy initialization, event synchronization.

import { useState, useEffect, useCallback } from 'react';

export function useLocalStorage<T>(key: string, initialValue: T) {
  // Lazy initializer: the function passed to useState runs only on first render.
  // This avoids reading localStorage on every re-render (it's synchronous I/O).
  const [storedValue, setStoredValue] = useState<T>(() => {
    if (typeof window === 'undefined') return initialValue; // SSR safety
    try {
      const item = window.localStorage.getItem(key);
      return item ? (JSON.parse(item) as T) : initialValue;
    } catch (error) {
      console.warn(`Error reading localStorage key "${key}":`, error);
      return initialValue;
    }
  });

  // Memoized setter that also persists to localStorage.
  const setValue = useCallback(
    (value: T | ((prev: T) => T)) => {
      setStoredValue((prev) => {
        // Support functional updater pattern (same as useState)
        const nextValue = value instanceof Function ? value(prev) : value;
        // Persist to localStorage
        try {
          window.localStorage.setItem(key, JSON.stringify(nextValue));
        } catch (error) {
          console.warn(`Error writing localStorage key "${key}":`, error);
        }
        return nextValue;
      });
    },
    [key]
  );

  // Sync across tabs: listen for storage events from other windows.
  useEffect(() => {
    function handleStorageChange(e: StorageEvent) {
      if (e.key === key && e.newValue !== null) {
        setStoredValue(JSON.parse(e.newValue) as T);
      }
    }
    window.addEventListener('storage', handleStorageChange);
    return () => window.removeEventListener('storage', handleStorageChange);
  }, [key]);

  return [storedValue, setValue] as const;
}

// ─── Usage ───────────────────────────────────────────────────────
// const [theme, setTheme] = useLocalStorage<'light' | 'dark'>('theme', 'light');
// setTheme('dark');           // Persists to localStorage immediately
// setTheme(prev => prev === 'light' ? 'dark' : 'light'); // Functional updater
```

## 4. Zustand Store with Middleware

```tsx
// store/appStore.ts
// ─────────────────────────────────────────────────────────────────
// Production Zustand store with:
// - TypeScript interface
// - persist middleware (localStorage)
// - devtools middleware (Redux DevTools integration)
// - immer middleware (mutable-style updates)

import { create } from 'zustand';
import { persist, devtools } from 'zustand/middleware';
import { immer } from 'zustand/middleware/immer';

// ─── Type definitions ────────────────────────────────────────────
interface CartItem {
  id: string;
  name: string;
  price: number;
  quantity: number;
}

interface AppState {
  // State slices
  cart: CartItem[];
  isCartOpen: boolean;

  // Computed (derived via selectors, not stored)
  // Actions
  addToCart: (item: Omit<CartItem, 'quantity'>) => void;
  removeFromCart: (id: string) => void;
  updateQuantity: (id: string, quantity: number) => void;
  toggleCart: () => void;
  clearCart: () => void;
}

// ─── Store creation ──────────────────────────────────────────────
// Middleware is applied inside-out: immer → devtools → persist → create
export const useAppStore = create<AppState>()(
  devtools(
    persist(
      immer((set) => ({
        // Initial state
        cart: [],
        isCartOpen: false,

        // Actions using immer: write "mutable" code, immer produces immutable updates
        addToCart: (item) =>
          set((state) => {
            const existing = state.cart.find((i) => i.id === item.id);
            if (existing) {
              existing.quantity += 1; // Looks mutable, but immer handles immutability
            } else {
              state.cart.push({ ...item, quantity: 1 });
            }
          }),

        removeFromCart: (id) =>
          set((state) => {
            state.cart = state.cart.filter((i) => i.id !== id);
          }),

        updateQuantity: (id, quantity) =>
          set((state) => {
            const item = state.cart.find((i) => i.id === id);
            if (item) item.quantity = Math.max(0, quantity);
          }),

        toggleCart: () =>
          set((state) => {
            state.isCartOpen = !state.isCartOpen;
          }),

        clearCart: () =>
          set((state) => {
            state.cart = [];
          }),
      })),
      {
        name: 'app-cart-storage', // localStorage key
        partialize: (state) => ({ cart: state.cart }), // Only persist cart, not UI state
      }
    ),
    { name: 'AppStore' } // Redux DevTools label
  )
);

// ─── Selectors (for fine-grained subscriptions) ──────────────────
// Each selector creates an independent subscription.
// Components using these only re-render when their specific value changes.
export const useCartItems = () => useAppStore((s) => s.cart);
export const useCartCount = () => useAppStore((s) => s.cart.reduce((sum, i) => sum + i.quantity, 0));
export const useCartTotal = () => useAppStore((s) => s.cart.reduce((sum, i) => sum + i.price * i.quantity, 0));
export const useIsCartOpen = () => useAppStore((s) => s.isCartOpen);
```

## 5. Server Action with Form Validation

```tsx
// app/contact/actions.ts
'use server';
// ─────────────────────────────────────────────────────────────────
// Server Actions run exclusively on the server.
// They can be called from Client Components via form action or direct invocation.
// They replace API routes for mutations (POST/PUT/DELETE).

import { z } from 'zod';
import { revalidatePath } from 'next/cache';

// Zod schema: runtime validation for the form data
const ContactSchema = z.object({
  name: z.string().min(2, 'Name must be at least 2 characters'),
  email: z.string().email('Invalid email address'),
  message: z.string().min(10, 'Message must be at least 10 characters'),
});

// Return type for the action (used by useFormState on the client)
export type ContactFormState = {
  success: boolean;
  errors?: Record<string, string[]>;
  message?: string;
};

// The Server Action function
export async function submitContact(
  prevState: ContactFormState,
  formData: FormData
): Promise<ContactFormState> {
  // Parse and validate form data
  const parsed = ContactSchema.safeParse({
    name: formData.get('name'),
    email: formData.get('email'),
    message: formData.get('message'),
  });

  // Return validation errors to the client
  if (!parsed.success) {
    return {
      success: false,
      errors: parsed.error.flatten().fieldErrors,
    };
  }

  // Server-side logic: save to database, send email, etc.
  try {
    await db.contact.create({ data: parsed.data });
    // Revalidate the page cache so new data appears
    revalidatePath('/contact');
    return { success: true, message: 'Message sent successfully!' };
  } catch {
    return { success: false, message: 'Failed to send. Please try again.' };
  }
}
```

```tsx
// app/contact/page.tsx
'use client';
import { useFormState, useFormStatus } from 'react-dom';
import { submitContact, type ContactFormState } from './actions';

const initialState: ContactFormState = { success: false };

export default function ContactPage() {
  // useFormState connects the Server Action to component state.
  // state updates when the action returns; action receives prevState + formData.
  const [state, formAction] = useFormState(submitContact, initialState);

  return (
    <form action={formAction} className="space-y-4 max-w-md">
      <div>
        <label htmlFor="name">Name</label>
        <input id="name" name="name" required className="block w-full border p-2" />
        {state.errors?.name && (
          <p className="text-red-500 text-sm">{state.errors.name[0]}</p>
        )}
      </div>

      <div>
        <label htmlFor="email">Email</label>
        <input id="email" name="email" type="email" required className="block w-full border p-2" />
        {state.errors?.email && (
          <p className="text-red-500 text-sm">{state.errors.email[0]}</p>
        )}
      </div>

      <div>
        <label htmlFor="message">Message</label>
        <textarea id="message" name="message" rows={4} required className="block w-full border p-2" />
        {state.errors?.message && (
          <p className="text-red-500 text-sm">{state.errors.message[0]}</p>
        )}
      </div>

      <SubmitButton />
      {state.success && <p className="text-green-600">{state.message}</p>}
    </form>
  );
}

// Separate component to use useFormStatus (must be a child of <form>)
function SubmitButton() {
  const { pending } = useFormStatus();
  return (
    <button type="submit" disabled={pending} className="bg-blue-600 text-white px-4 py-2 rounded">
      {pending ? 'Sending...' : 'Send Message'}
    </button>
  );
}
```

## 6. Middleware & Route Protection

```tsx
// middleware.ts (root of project)
// ─────────────────────────────────────────────────────────────────
// Next.js Middleware runs on the Edge Runtime BEFORE the request reaches
// any page or API route. Use it for: auth checks, redirects, headers, rate limiting.

import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Read auth token from cookies
  const token = request.cookies.get('auth-token')?.value;

  // Protected routes: redirect to login if no token
  if (pathname.startsWith('/dashboard') && !token) {
    const loginUrl = new URL('/login', request.url);
    // Preserve the original URL so we can redirect back after login
    loginUrl.searchParams.set('callbackUrl', pathname);
    return NextResponse.redirect(loginUrl);
  }

  // Add security headers to all responses
  const response = NextResponse.next();
  response.headers.set('X-Frame-Options', 'DENY');
  response.headers.set('X-Content-Type-Options', 'nosniff');
  response.headers.set('Referrer-Policy', 'strict-origin-when-cross-origin');

  return response;
}

// Config: specify which routes this middleware applies to.
// Excludes static files and API routes for performance.
export const config = {
  matcher: [
    // Match all routes except static files and Next.js internals
    '/((?!_next/static|_next/image|favicon.ico|api/).*)',
  ],
};
```
